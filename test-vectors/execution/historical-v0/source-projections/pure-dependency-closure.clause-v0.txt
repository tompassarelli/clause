relation build/imports
  reads {consumer: Component} imports {dependency: Component}
  subject consumer
  mode given consumer yields dependency: many

relation impact/depends
  reads {consumer: Component} depends on {dependency: Component}
  subject consumer
  mode given consumer yields dependency: many

law impact/direct-dependency
  if
    ?consumer imports ?dependency
  then
    ?consumer depends on ?dependency

law impact/transitive-dependency
  if
    ?consumer depends on ?middle
    ?middle depends on ?dependency
  then
    ?consumer depends on ?dependency

derive impact/direct-dependency
derive impact/transitive-dependency
